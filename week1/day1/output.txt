    CREATE TABLE Department (
        department_id INT PRIMARY KEY,
        name VARCHAR(50)
    );
    
    CREATE TABLE Employee (
        emp_id INT PRIMARY KEY,
        name VARCHAR(100),
        age INT,
        salary DECIMAL(10,2),
        department_id INT,
        hire_date DATE
    );
    
    CREATE TABLE Project (
        project_id INT PRIMARY KEY,
        name VARCHAR(100),
        department_id INT
    );
